cmake_minimum_required(VERSION 3.20)
project(qtbench LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O2 -DNDEBUG")
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Quick)
set(CORE_DIR /home/user/eval/core)
add_executable(qtbench main.cpp model.h canvas.h common.h qml.qrc)
target_include_directories(qtbench PRIVATE ${CORE_DIR})
target_link_libraries(qtbench PRIVATE Qt6::Quick ${CORE_DIR}/target/release/libbenchcore.a pthread dl m)
